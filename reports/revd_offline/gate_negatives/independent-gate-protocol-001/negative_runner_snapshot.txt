"""Independent, read-only fault injection against the offline evidence gate.

Only this runner's fresh evidence directory and follow-up review are written.
JSON and text alterations stay in memory; accepted RTL artifacts are untouched.
These are software gate negatives, not additional RTL mutation experiments.
"""
from __future__ import annotations

import argparse
from contextlib import ExitStack
import copy
import hashlib
import json
import os
from pathlib import Path
import sys
import time
import types
from unittest.mock import patch

sys.dont_write_bytecode = True
ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))
from scripts.revd_offline_common import put, read_json, run_id, sha, utc


def bound_path(value):
    p = Path(value)
    return p if p.is_absolute() else ROOT / p


def identity(path):
    path = Path(path)
    return {"path": str(path.relative_to(ROOT)), "sha256": sha(path),
            "bytes": path.stat().st_size}


def compact(result):
    if not isinstance(result, dict):
        return result
    return {k: v for k, v in result.items() if k not in ("results", "config")}


class Checks:
    def __init__(self, gate):
        self.gate = gate
        self.results = []

    def run(self, name, function, *, reject=False, fragments=(), json_overrides=None,
            text_overrides=None):
        """A negative passes only if the real gate throws an expected ValueError."""
        started = time.monotonic()
        original_json = self.gate.read_json
        original_text = Path.read_text
        json_overrides = {str(Path(p)): v for p, v in (json_overrides or {}).items()}
        text_overrides = {str(Path(p)): v for p, v in (text_overrides or {}).items()}

        def json_read(path):
            key = str(Path(path))
            return copy.deepcopy(json_overrides[key]) if key in json_overrides else original_json(path)

        def text_read(path, *args, **kwargs):
            key = str(Path(path))
            value = original_text(path, *args, **kwargs)
            return text_overrides[key](value) if key in text_overrides else value

        record = {"name": name, "expected": "REJECT" if reject else "ACCEPT"}
        with ExitStack() as stack:
            if json_overrides:
                stack.enter_context(patch.object(self.gate, "read_json", json_read))
            if text_overrides:
                stack.enter_context(patch.object(Path, "read_text", text_read))
            try:
                result = function()
            except ValueError as exc:
                record.update(actual="REJECT", error=str(exc))
                record["passed"] = reject and (
                    not fragments or any(fragment in str(exc) for fragment in fragments))
            except Exception as exc:
                record.update(actual="UNEXPECTED_EXCEPTION", exception=type(exc).__name__,
                              error=str(exc), passed=False)
            else:
                record.update(actual="ACCEPT", result=compact(result), passed=not reject)
        record["seconds"] = time.monotonic() - started
        self.results.append(record)
        print(json.dumps(record), flush=True)
        return record["passed"]


def changed_manifest(gate, manifest, mutate, command_index=None):
    value = copy.deepcopy(gate.read_json(manifest))
    mutate(value)
    overrides = {manifest: value}
    if command_index is not None:
        cmd = value["commands"][command_index]
        log = ROOT / cmd["log"]
        overrides[log.with_suffix(log.suffix + ".command.json")] = cmd
    return overrides


def protocol_negatives(checks, protocol):
    gate = checks.gate
    summary = gate.read_json(protocol)
    entries = {t["name"]: t for t in summary["tests"]}
    manifest = ROOT / entries["last_fire_invalid_l1"]["manifest"]
    check = lambda: gate.check_case(manifest)
    checks.run("positive_current_case", check)
    if not checks.run("positive_complete_protocol", lambda: gate.check_protocol(protocol)):
        return manifest

    for index, name in ((0, "compile_nonzero"), (1, "simulation_nonzero")):
        overrides = changed_manifest(gate, manifest,
                                     lambda r, i=index: r["commands"][i].update(exit_code=1))
        checks.run(name, check, reject=True, fragments=("nonzero real subprocess exit",),
                   json_overrides=overrides)

    overrides = changed_manifest(
        gate, manifest, lambda r: r["commands"][1].update(argv=["/usr/bin/true"]), 1)
    checks.run("fake_true_simulator_argv", check, reject=True,
               fragments=("simulation did not execute bound",), json_overrides=overrides)

    def omit_selected_rtl(value):
        value["source_sha256"].pop("rtl/revd_opt/core_stream_frontend_opt.sv")
        value["source_sha256"]["./numeric/fixedpoint.py"] = value["source_sha256"]["numeric/fixedpoint.py"]

    checks.run("delete_RTL_replace_with_path_alias", check, reject=True,
               fragments=("incomplete/aliased source inventory",),
               json_overrides=changed_manifest(gate, manifest, omit_selected_rtl))

    output = manifest.parent / "output.hex"

    def alter_output(text, mode):
        lines = text.splitlines()
        if mode == "delete_last_output":
            lines = lines[:-1]
        else:
            fields = lines[0].split()
            if mode == "wrong_index":
                fields[1] = "00000001" if fields[1] != "00000001" else "00000002"
            else:
                position = 3 if mode == "wrong_overflow_flag" else 4
                fields[position] = "1" if fields[position] == "0" else "0"
            lines[0] = " ".join(fields)
        return "\n".join(lines) + "\n"

    for name in ("wrong_overflow_flag", "wrong_saturated_flag", "wrong_index", "delete_last_output"):
        checks.run(name, check, reject=True,
                   fragments=("output/flags/index/last mismatch", "missing/extra output"),
                   text_overrides={output: lambda text, name=name: alter_output(text, name)})

    changed = copy.deepcopy(gate.read_json(manifest))
    changed["config"]["PM"] = 2
    checks.run("coherent_json_but_mixed_PM_compile", check, reject=True,
               fragments=("compile parameter differs from config PM",),
               json_overrides={manifest: changed, manifest.parent / "config.json": changed["config"]})

    missing = copy.deepcopy(summary)
    missing["tests"] = [t for t in missing["tests"] if t["name"] != "pending_last_blocked_l2"]
    checks.run("missing_terminal_boundary_test", lambda: gate.check_protocol(protocol), reject=True,
               fragments=("missing/duplicate/unexpected protocol tests",),
               json_overrides={protocol: missing})

    aliased = copy.deepcopy(summary)
    for test in aliased["tests"]:
        if test["name"] == "pending_last_blocked_l2":
            test["manifest"] = entries["last_fire_invalid_l1"]["manifest"]
    checks.run("aliased_boundary_manifest", lambda: gate.check_protocol(protocol), reject=True,
               fragments=("aliased protocol manifests",), json_overrides={protocol: aliased})

    mutant_alias = copy.deepcopy(summary)
    source = Path(entries["MUT_DROP_K63"]["manifest"])
    # Different textual path to the same real bytes bypasses mere string uniqueness.
    alias = source.parent / ".." / source.parent.name / source.name
    for test in mutant_alias["tests"]:
        if test["name"] == "MUT_UNSIGNED":
            test["manifest"] = str(alias)
    checks.run("semantic_mutant_alias", lambda: gate.check_protocol(protocol), reject=True,
               fragments=("mutant identity alias", "aliased protocol manifests"),
               json_overrides={protocol: mutant_alias})

    stats = manifest.parent / "stats.txt"
    checks.run("missing_terminal_boundary_trace", lambda: gate.check_protocol(protocol), reject=True,
               fragments=("missing/mixed boundary trace",),
               text_overrides={stats: lambda text: "\n".join(
                   s for s in text.splitlines() if not s.startswith("BOUNDARY ")) + "\n"})
    reset_manifest = ROOT / entries["reset_l1_phase1"]["manifest"]
    checks.run("missing_reset_phase_trace", lambda: gate.check_protocol(protocol), reject=True,
               fragments=("reset phase trace missing",),
               text_overrides={reset_manifest.parent / "stats.txt": lambda text: "\n".join(
                   s for s in text.splitlines() if not s.startswith("RESET ")) + "\n"})
    return manifest


def frozen_negatives(checks, path):
    gate = checks.gate
    if not checks.run("positive_complete_frozen36", lambda: gate.check_suite(path, 36, 4)):
        return
    summary = gate.read_json(path)
    missing = copy.deepcopy(summary)
    missing["cases"] = missing["cases"][:-1]
    checks.run("missing_frozen_case", lambda: gate.check_suite(path, 36, 4), reject=True,
               fragments=("missing required cases", "full36 coverage/output count"),
               json_overrides={path: missing})
    alias = copy.deepcopy(summary)
    alias["cases"][-1]["manifest"] = alias["cases"][0]["manifest"]
    checks.run("aliased_frozen_case_manifest", lambda: gate.check_suite(path, 36, 4), reject=True,
               fragments=("duplicate case manifests",), json_overrides={path: alias})
    label = copy.deepcopy(summary)
    label["cases"][-1]["case"] = label["cases"][0]["case"]
    checks.run("aliased_frozen_sample_layer_identity", lambda: gate.check_suite(path, 36, 4), reject=True,
               fragments=("full36 sample/layer identity set mismatch",), json_overrides={path: label})


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--run-id")
    parser.add_argument("--protocol", default="reports/revd_offline/protocol/20261017T062232Z_d44560d2/summary.json")
    parser.add_argument("--full36", help="Add frozen36 checks only when its actual successful summary exists.")
    args = parser.parse_args()
    destination = ROOT / "reports/revd_offline/gate_negatives" / (args.run_id or run_id())
    destination.mkdir(parents=True, exist_ok=False)
    started = utc()
    gate_path = ROOT / "scripts/revd_offline_gate.py"
    gate_bytes = gate_path.read_bytes()
    gate = types.ModuleType("independent_gate_snapshot")
    gate.__file__ = str(gate_path)
    exec(compile(gate_bytes, str(gate_path), "exec"), gate.__dict__)
    (destination / "gate_source_snapshot.txt").write_bytes(gate_bytes)
    (destination / "negative_runner_snapshot.txt").write_bytes(Path(__file__).read_bytes())
    protocol = bound_path(args.protocol)
    sources = [gate_path, Path(__file__), ROOT / "scripts/revd_offline_common.py", protocol]
    if args.full36:
        sources.append(bound_path(args.full36))
    before = [identity(path) for path in sources]
    checks = Checks(gate)
    selected = protocol_negatives(checks, protocol)
    if args.full36:
        frozen_negatives(checks, bound_path(args.full36))
    changed = [item["path"] for item in before if sha(ROOT / item["path"]) != item["sha256"]]
    ok = bool(checks.results) and all(test["passed"] for test in checks.results) and not changed
    result = {
        "schema_version": 1, "status": "PASS" if ok else "FAIL", "started_utc": started,
        "finished_utc": utc(), "pid": os.getpid(),
        "run_identity": os.environ.get("KWT_OFFLINE_RUN_ID"),
        "scope": "Independent software evidence-gate negatives; not RTL mutation or contest acceptance.",
        "source_and_input_sha256": before, "changed_during_checks": changed,
        "tests": checks.results, "test_count": len(checks.results),
        "negative_count": sum(test["expected"] == "REJECT" for test in checks.results),
        "positive_count": sum(test["expected"] == "ACCEPT" for test in checks.results),
        "selected_current_case": identity(selected),
        "full36_checked": bool(args.full36) and any(
            test["name"] == "positive_complete_frozen36" and test["passed"] for test in checks.results),
        "CPU_checked": False, "fake_CPU_negative": "NOT_RUN_AWAITING_CPU_GATE",
        "mutation_policy": "Every negative must raise ValueError with its expected reason; unrelated exceptions fail.",
        "disk_mutation_policy": "No existing artifacts changed; only in-memory read_json/read_text replacement.",
        "prior_red_evidence": identity(ROOT / "work/revd_offline_20261017/review_verification_round1.json"),
        "parent_accepted": False, "official_contest_acceptance": False,
    }
    put(destination / "summary.json", result)
    followup = {
        "schema_version": 1, "review_round": "1_followup", "status": result["status"],
        "created_utc": utc(), "evidence": identity(destination / "summary.json"),
        "tests": result["test_count"], "software_gate_negatives": result["negative_count"],
        "positive_controls": result["positive_count"], "full36_checked": result["full36_checked"],
        "CPU_checked": False, "remaining": (["full36 exact-set negatives"] if not args.full36 else []) +
            ["CPU fake-done and final-writeback gate negatives", "final portable package review"],
        "review_scope_passed": ok, "final_reviews_passed": False, "parent_accepted": False,
        "official_contest_acceptance": False,
    }
    put(ROOT / "work/revd_offline_20261017/review_verification_round1_followup.json", followup)
    print(json.dumps({"evidence": str(destination / "summary.json"), "status": result["status"],
                      "tests": len(checks.results), "changed": changed}), flush=True)
    return 0 if ok else 1


if __name__ == "__main__":
    raise SystemExit(main())
